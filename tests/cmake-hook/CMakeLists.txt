cmake_minimum_required(VERSION 3.16)
project(libcore_hook_contract NONE)
set(IDF_TARGET esp32s3)
add_library(esp_wifi_core STATIC IMPORTED)
set_target_properties(esp_wifi_core PROPERTIES IMPORTED_LOCATION /unused/vendor/libcore.a)
add_library(test_libcore INTERFACE)
set_target_properties(test_libcore PROPERTIES OUTPUT_NAME core)
function(idf_component_get_property output component property)
    if(NOT component STREQUAL "libcore")
        message(FATAL_ERROR "Unexpected component: ${component}")
    endif()
    if(property STREQUAL "COMPONENT_LIB")
        set(${output} test_libcore PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected component property: ${property}")
    endif()
endfunction()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/replace_core.cmake")
get_target_property(actual esp_wifi_core IMPORTED_LOCATION)
set(expected "${CMAKE_CURRENT_BINARY_DIR}/libcore.a")
if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Unexpected replacement path: ${actual}")
endif()
get_target_property(dependencies esp_wifi_core MANUALLY_ADDED_DEPENDENCIES)
if(dependencies)
    message(FATAL_ERROR "The hook must not add imported-target dependencies")
endif()
message(STATUS "PASS: replacement archive path and absence of manual dependencies")
